name: Update stopwatch snapshot

on:
  schedule:
    - cron: "*/5 * * * *"   # every 5 minutes (GitHub's minimum)
  workflow_dispatch:         # lets you run it by hand

permissions:
  contents: write

jobs:
  update:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-node@v4
        with:
          node-version: 20
      - run: node update-readme.js
      - name: Commit if changed
        run: |
          git config user.name "stopwatch-bot"
          git config user.email "stopwatch-bot@users.noreply.github.com"
          git add README.md
          git diff --cached --quiet || (git commit -m "Update stopwatch snapshot" && git push)
